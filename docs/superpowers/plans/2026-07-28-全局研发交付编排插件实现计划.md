# 全局研发交付编排 Plugin 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 构建一个面向 Codex CLI 的全局个人 Plugin，将已注册项目的需求从方案批准自动推进到 `READY_TO_MERGE`，并以 Python、SQLite、Git/GitHub 和 Windows Task Scheduler 提供可恢复、可审计的执行能力。

**架构：** `$change-delivery` 负责人与 Codex 的交互和 sub-agent 编排，Python CLI 负责确定性状态迁移、幂等和外部事实对账。运行状态保存在用户目录的 SQLite 与 run 产物中，项目差异由全局注册表表达；第一版使用 ExampleProject 后端验证 Draft PR、共享 `test` 分支部署、SHA 验收和 Review 修复闭环。

**技术栈：** Python 3.12、Pydantic 2、PyYAML、SQLite、filelock、pytest、Git、GitHub CLI、Codex Skills、Windows PowerShell/Task Scheduler

```python
from pydantic import ValidationError

from change_delivery.domain import CommandSpec, RunState


def test_command_spec_rejects_shell_executable() -> None:
    with pytest.raises(ValidationError):
        CommandSpec(
            executable="powershell",
            args=("-Command", "Get-Date"),
            workdir=Path("C:/repo"),
        )


def test_command_spec_preserves_literal_argv_with_shell_disabled() -> None:
    spec = CommandSpec(
        executable="python",
        args=("--expression", "cmd /c echo; value|literal"),
        workdir=Path("C:/repo"),
    )
    assert spec.args == ("--expression", "cmd /c echo; value|literal")


def test_run_state_has_ready_to_merge_terminal_gate() -> None:
    assert RunState.READY_TO_MERGE.value == "READY_TO_MERGE"
    assert RunState.COMPLETED.value == "COMPLETED"


def test_command_spec_resolves_workdir() -> None:
    spec = CommandSpec(executable="python", args=("-m", "pytest"), workdir=Path("C:/repo"))
    assert spec.workdir == Path("C:/repo")
```

`CommandSpec.executable` 还必须按大小写无关的 basename 拒绝 PowerShell、cmd、
bash、sh、wsl 和删除工具，并拒绝 `.ps1/.bat/.cmd/.sh` 脚本后缀；此限制只针对
可执行文件，不扫描普通参数中的 Shell 文本。

另以参数化测试覆盖项目预检 profile：直接执行仅允许 `pytest`、`ruff`、`mypy`；
`python`、`python3`、`py` 仅允许 `-m pytest/ruff/mypy/compileall`。必须拒绝
`python -c`、Python 脚本、`git clean`、`gh`、`codex`、`node` 和未知工具。
此 profile 有意保持严格：不接受 `py -3.12`、`python -I` 等启动器参数；`ruff`
必须以 `check` 为首个工具参数且显式包含 `--no-fix`，并拒绝 `--fix`、
`--fix-only`、`--unsafe-fixes`、`--add-noqa`；允许 `--config fix=true` 仅因为
显式 `--no-fix` 在本 profile 中覆盖配置启用的修复；
严格 profile 还会有意拒绝独立参数终止符 `--`，避免其后的 `--no-fix` 被 Ruff
解释为位置参数而绕过只读门禁；
`mypy` 拒绝 `--install-types`；`pytest` 拒绝 `--cache-clear` 和 `--basetemp`。
`compileall` 会写入 `__pycache__`，执行器必须在隔离 worktree 中运行预检。
`CommandSpec` 只描述 `shell=False` 的结构化 argv；后续内部 Git 适配器不得从项目 YAML
读取任意 Git 命令。

在 `src/change_delivery/domain.py` 定义：

```python
from enum import StrEnum
from pathlib import Path

from pydantic import BaseModel, ConfigDict, Field, field_validator, model_validator


class RunState(StrEnum):
    DRAFT = "DRAFT"
    AWAITING_DESIGN_APPROVAL = "AWAITING_DESIGN_APPROVAL"
    APPROVED = "APPROVED"
    PREPARING_WORKTREE = "PREPARING_WORKTREE"
    IMPLEMENTING = "IMPLEMENTING"
    LOCAL_VALIDATION = "LOCAL_VALIDATION"
    INTERNAL_REVIEW = "INTERNAL_REVIEW"
    COMMITTING = "COMMITTING"
    PUSHING_FEATURE = "PUSHING_FEATURE"
    PR_DRAFT = "PR_DRAFT"
    DEPLOYING_TEST = "DEPLOYING_TEST"
    ACCEPTANCE_TESTING = "ACCEPTANCE_TESTING"
    PR_READY_FOR_REVIEW = "PR_READY_FOR_REVIEW"
    MONITORING_CI_REVIEW = "MONITORING_CI_REVIEW"
    FIXING_CI_REVIEW = "FIXING_CI_REVIEW"
    READY_TO_MERGE = "READY_TO_MERGE"
    MERGED = "MERGED"
    CLEANING_LOCAL = "CLEANING_LOCAL"
    COMPLETED = "COMPLETED"
    BLOCKED = "BLOCKED"
    ENVIRONMENT_PREEMPTED = "ENVIRONMENT_PREEMPTED"
    REVIEW_DECISION_REQUIRED = "REVIEW_DECISION_REQUIRED"
    FAILED_TERMINAL = "FAILED_TERMINAL"
    CANCELLED = "CANCELLED"


class CommandSpec(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)
    executable: str = Field(min_length=1)
    args: tuple[str, ...] = Field(default_factory=tuple)
    workdir: Path
    timeout_seconds: int = Field(default=300, ge=1, le=7200)
    env_names: tuple[str, ...] = Field(default_factory=tuple)

    @field_validator("args")
    @classmethod
    def reject_multiline_arguments(cls, args: tuple[str, ...]) -> tuple[str, ...]:
        if any("\r" in arg or "\n" in arg for arg in args):
            raise ValueError("参数中禁止换行")
        return args


class ProjectConfig(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)
    project_id: str = Field(pattern=r"^[a-z0-9][a-z0-9-]{1,62}$")
    repo_path: Path
    remote: str = Field(pattern=r"^[A-Za-z0-9._-]+$")
    base_branch: str
    test_branch: str
    github_repository: str = Field(pattern=r"^[^/]+/[^/]+$")
    version_endpoint: str
    version_sha_field: str
    required_pre_pr_checks: tuple[CommandSpec, ...] = Field(min_length=1)
    credentials: tuple[str, ...] = Field(default_factory=tuple)
    production_enabled: bool = False

    @model_validator(mode="after")
    def require_validation_profile_checks(self) -> "ProjectConfig":
        direct_tools = {"pytest", "ruff", "mypy"}
        python_tools = {"python", "python3", "py"}
        python_modules = {"pytest", "ruff", "mypy", "compileall"}
        for check in self.required_pre_pr_checks:
            executable = check.executable.replace("\\", "/").rsplit("/", 1)[-1].lower()
            if executable.endswith(".exe"):
                executable = executable[:-4]
            direct = executable in direct_tools
            module = (
                executable in python_tools
                and len(check.args) >= 2
                and check.args[0] == "-m"
                and check.args[1] in python_modules
            )
            if not direct and not module:
                raise ValueError("项目预检命令不属于第一版验证工具白名单")
        return self


class LoadedProjectConfig(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)
    config: ProjectConfig
    config_hash: str
    source: Path
```

增加 `RunRecord`、`EventRecord` Pydantic 模型，并按以下 SQL 边界实现：

```python
class RunRepository:
    def __init__(self, database: Database) -> None:
        self.database = database

    def create_run(self, run_id: str, project_id: str, config_hash: str) -> None:
        now = datetime.now(timezone.utc).isoformat()
        with self.database.transaction() as connection:
            connection.execute(
                """
                INSERT INTO runs(run_id, project_id, config_hash, state, created_at, updated_at)
                VALUES (?, ?, ?, ?, ?, ?)
                """,
                (run_id, project_id, config_hash, RunState.DRAFT.value, now, now),
            )

    def get_run(self, run_id: str) -> RunRecord:
        with self.database.connection() as connection:
            row = connection.execute(
                "SELECT * FROM runs WHERE run_id = ?",
                (run_id,),
            ).fetchone()
        if row is None:
            raise KeyError(run_id)
        return RunRecord.model_validate(dict(row))

    def list_active_runs(self) -> list[RunRecord]:
        terminal = (
            RunState.COMPLETED.value,
            RunState.CANCELLED.value,
            RunState.FAILED_TERMINAL.value,
        )
        with self.database.connection() as connection:
            rows = connection.execute(
                "SELECT * FROM runs WHERE state NOT IN (?, ?, ?) ORDER BY created_at",
                terminal,
            ).fetchall()
        return [RunRecord.model_validate(dict(row)) for row in rows]

    def transition(
        self,
        run_id: str,
        expected: RunState,
        target: RunState,
        event_type: str,
        evidence_ids: list[str],
    ) -> None:
        now = datetime.now(timezone.utc).isoformat()
        with self.database.transaction() as connection:
            row = connection.execute(
                "SELECT state FROM runs WHERE run_id = ?",
                (run_id,),
            ).fetchone()
            if row is None:
                raise KeyError(run_id)
            if row["state"] != expected.value:
                raise RuntimeError(
                    f"状态已变化：expected={expected.value}, actual={row['state']}"
                )
            connection.execute(
                "UPDATE runs SET state = ?, updated_at = ? WHERE run_id = ?",
                (target.value, now, run_id),
            )
            connection.execute(
                """
                INSERT INTO events(
                    event_id, run_id, event_type, from_state, to_state,
                    evidence_json, created_at
                )
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    str(uuid4()),
                    run_id,
                    event_type,
                    expected.value,
                    target.value,
                    json.dumps(evidence_ids),
                    now,
                ),
            )

    def list_events(self, run_id: str) -> list[EventRecord]:
        with self.database.connection() as connection:
            rows = connection.execute(
                "SELECT * FROM events WHERE run_id = ? ORDER BY created_at",
                (run_id,),
            ).fetchall()
        return [EventRecord.model_validate(dict(row)) for row in rows]
```

`transition` 必须在同一事务中验证 `expected`、更新 `runs` 并插入 `events`。`Database.connection()` 和 `transaction()` 都要设置 `row_factory = sqlite3.Row`。

```python
SENSITIVE_PATTERN = re.compile(
    r"(?i)(authorization:\s*bearer\s+|cookie:\s*|password=|token=)([^\s&]+)"
)


def redact_text(text: str, secrets: Mapping[str, str]) -> str:
    result = text
    for value in sorted((value for value in secrets.values() if value), key=len, reverse=True):
        result = result.replace(value, "***")
    return SENSITIVE_PATTERN.sub(lambda match: f"{match.group(1)}***", result)
```
