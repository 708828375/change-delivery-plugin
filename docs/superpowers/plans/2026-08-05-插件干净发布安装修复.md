# 插件干净发布安装修复实施计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 将个人插件安装从“Junction 指向完整开发仓库”改为“原子发布只含 `.codex-plugin` 与 `skills` 的普通目录”，消除 Codex 递归复制测试临时目录时的访问拒绝。

**架构：** 新增一个只定义发布函数的 PowerShell 辅助脚本，负责构建白名单临时目录、验证目录树无重解析点，并安全切换最终发布目录。现有安装器负责固定信任锚、调用官方插件校验器、调用发布函数和执行 `codex plugin add`；Python editable package 保持不变。

**技术栈：** Windows PowerShell 5.1、Python 3.12、pytest、Codex CLI 0.146.0、NTFS 普通目录与 Junction 迁移。

---

from pathlib import Path
import subprocess


ROOT = Path(__file__).parents[2]
PUBLISHER = ROOT / "scripts" / "plugin-publisher.ps1"


def run_ps(script: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [
            "powershell.exe",
            "-NoProfile",
            "-NonInteractive",
            "-ExecutionPolicy",
            "Bypass",
            "-Command",
            script,
        ],
        capture_output=True,
        text=True,
        check=False,
    )


def test_clean_stage_contains_only_plugin_runtime_entries(tmp_path: Path) -> None:
    source = tmp_path / "source"
    plugins = tmp_path / "plugins"
    (source / ".codex-plugin").mkdir(parents=True)
    (source / "skills" / "change-delivery").mkdir(parents=True)
    (source / ".codex-plugin" / "plugin.json").write_text("{}", encoding="utf-8")
    (source / "skills" / "change-delivery" / "SKILL.md").write_text("skill", encoding="utf-8")
    (source / ".git").mkdir()
    (source / ".pytest-denied").mkdir()
    plugins.mkdir()

    command = rf"""
. '{PUBLISHER}'
$stage = New-CleanPluginStage '{source}' '{plugins}'
$names = @((Get-ChildItem -LiteralPath $stage -Force).Name | Sort-Object)
if (($names -join ',') -cne '.codex-plugin,skills') {{ exit 31 }}
if (Test-Path -LiteralPath (Join-Path $stage '.git')) {{ exit 32 }}
if (Test-Path -LiteralPath (Join-Path $stage '.pytest-denied')) {{ exit 33 }}
Assert-ClosedPluginReleaseTree $stage
"""
    result = run_ps(command)
    assert result.returncode == 0, result.stderr + result.stdout

- [ ] **步骤 3：实现最小白名单发布函数**

创建 `scripts/plugin-publisher.ps1`，实现以下接口：

```powershell
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

function Assert-OrdinaryDirectory([string] $Path, [string] $Label) {
    $item = Get-Item -LiteralPath $Path -Force -ErrorAction Stop
    if (
        $item -isnot [IO.DirectoryInfo] -or
        ($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 -or
        $null -ne $item.LinkType
    ) {
        throw "$Label must be an ordinary directory."
    }
}

function Assert-ClosedPluginReleaseTree([string] $Path) {
    Assert-OrdinaryDirectory $Path 'Plugin release root'
    $allowed = @('.codex-plugin', 'skills')
    $entries = @(Get-ChildItem -LiteralPath $Path -Force -ErrorAction Stop)
    if (@($entries | Where-Object { $_.Name -cnotin $allowed }).Count -ne 0) {
        throw 'Plugin release contains an unexpected top-level entry.'
    }
    foreach ($required in $allowed) {
        $requiredPath = Join-Path $Path $required
        Assert-OrdinaryDirectory $requiredPath "Plugin release entry $required"
    }
    foreach ($entry in @(Get-ChildItem -LiteralPath $Path -Recurse -Force -ErrorAction Stop)) {
        if (
            ($entry.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 -or
            $null -ne $entry.LinkType
        ) {
            throw 'Plugin release tree contains a link or reparse point.'
        }
    }
}

function New-CleanPluginStage([string] $SourceRoot, [string] $PluginsRoot) {
    Assert-OrdinaryDirectory $SourceRoot 'Plugin source root'
    Assert-OrdinaryDirectory (Join-Path $SourceRoot '.codex-plugin') 'Plugin manifest directory'
    Assert-OrdinaryDirectory (Join-Path $SourceRoot 'skills') 'Plugin skills directory'
    Assert-OrdinaryDirectory $PluginsRoot 'Plugins root'
    $stage = Join-Path $PluginsRoot ('.change-delivery-plugin.installing-' + [guid]::NewGuid().ToString('N'))
    New-Item -ItemType Directory -Path $stage -ErrorAction Stop | Out-Null
    try {
        Copy-Item -LiteralPath (Join-Path $SourceRoot '.codex-plugin') -Destination $stage -Recurse -ErrorAction Stop
        Copy-Item -LiteralPath (Join-Path $SourceRoot 'skills') -Destination $stage -Recurse -ErrorAction Stop
        Assert-ClosedPluginReleaseTree $stage
        return $stage
    }
    catch {
        if (Test-Path -LiteralPath $stage) {
            [IO.Directory]::Delete($stage, $true)
        }
        throw
    }
}
```

- [ ] **步骤 3：实现受控切换与回滚**

在 `scripts/plugin-publisher.ps1` 增加固定插件身份检查：

```powershell
function Assert-ManagedPluginRelease([string] $Path) {
    Assert-ClosedPluginReleaseTree $Path
    $manifestPath = Join-Path $Path '.codex-plugin\plugin.json'
    $manifest = ConvertFrom-Json (
        [IO.File]::ReadAllText($manifestPath, [Text.UTF8Encoding]::new($false, $true))
    )
    if (
        $manifest -isnot [PSCustomObject] -or
        $null -eq $manifest.name -or
        [string] $manifest.name -cne 'change-delivery-plugin'
    ) {
        throw 'Managed plugin release has an unexpected identity.'
    }
}
```

然后增加原子切换函数：

```powershell
function Install-CleanPluginStage {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [string] $StageRoot,
        [Parameter(Mandatory)] [string] $TargetRoot,
        [Parameter(Mandatory)] [string] $ExpectedLegacySource
    )

    Assert-ManagedPluginRelease $StageRoot
    $parent = Split-Path -Parent $TargetRoot
    Assert-OrdinaryDirectory $parent 'Plugins root'
    $backup = Join-Path $parent ('.change-delivery-plugin.backup-' + [guid]::NewGuid().ToString('N'))
    $legacyTarget = $null

    if (Test-Path -LiteralPath $TargetRoot) {
        $current = Get-Item -LiteralPath $TargetRoot -Force -ErrorAction Stop
        if ($current.LinkType -eq 'Junction') {
            $targets = @($current.Target)
            if ($targets.Count -ne 1) { throw 'Legacy plugin junction target is ambiguous.' }
            $legacyTarget = (Resolve-Path -LiteralPath ([string] $targets[0]) -ErrorAction Stop).ProviderPath
            $expected = (Resolve-Path -LiteralPath $ExpectedLegacySource -ErrorAction Stop).ProviderPath
            if ($legacyTarget.TrimEnd('\') -cne $expected.TrimEnd('\')) {
                throw 'Legacy plugin junction points to an unexpected source.'
            }
            $current.Delete($false)
        }
        else {
            Assert-ManagedPluginRelease $TargetRoot
            Move-Item -LiteralPath $TargetRoot -Destination $backup -ErrorAction Stop
        }
    }

    try {
        Move-Item -LiteralPath $StageRoot -Destination $TargetRoot -ErrorAction Stop
    }
    catch {
        if (Test-Path -LiteralPath $backup) {
            Move-Item -LiteralPath $backup -Destination $TargetRoot -ErrorAction Stop
        }
        elseif ($null -ne $legacyTarget -and -not (Test-Path -LiteralPath $TargetRoot)) {
            New-Item -ItemType Junction -Path $TargetRoot -Target $legacyTarget -ErrorAction Stop | Out-Null
        }
        throw
    }

    if (Test-Path -LiteralPath $backup) {
        Assert-ManagedPluginRelease $backup
        [IO.Directory]::Delete($backup, $true)
    }
}
```

同时增加 `Assert-ManagedPluginRelease`：先调用 `Assert-ClosedPluginReleaseTree`，再严格读取 `.codex-plugin/plugin.json`，要求根对象的 `name` 精确等于 `change-delivery-plugin`。普通目录更新只接受通过该检查的固定目标路径；不在发布目录中添加额外管理文件。
